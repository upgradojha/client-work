import os import pandas as pd import easyocr from google.colab import files from pdf2image import convert_from_path from IPython.display import clear_output # Initialize EasyOCR (uses PyTorch under the hood, bypasses Paddle C++ errors) reader = easyocr.Reader(['en'], gpu=False) print("Select multiple images or a single PDF to process:") uploaded_files = files.upload() clear_output(wait=True) results = [] for filename in uploaded_files.keys(): print(f"Processing: {filename}") if filename.lower().endswith('.pdf'): pages = convert_from_path(filename, dpi=300) for page_num, page_image in enumerate(pages, start=1): temp_image_path = f"temp_{page_num}.png" page_image.save(temp_image_path, 'PNG') # detail=0 directly returns extracted lines as text strings page_text = reader.readtext(temp_image_path, detail=0) results.append({ "Source File": filename, "Page/Image": f"Page {page_num}", "Extracted Text": "\n".join(page_text) }) os.remove(temp_image_path) print(f" -> Finished Page {page_num}") elif filename.lower().endswith(('.png', '.jpg', '.jpeg', '.bmp', '.tiff')): page_text = reader.readtext(filename, detail=0) results.append({ "Source File": filename, "Page/Image": "Image", "Extracted Text": "\n".join(page_text) }) print(f" -> Finished {filename}") if results: output_filename = "OCR_Extracted_Data.xlsx" df = pd.DataFrame(results) df.to_excel(output_filename, index=False) print(f"\n✅ Processing complete. Generated {output_filename} with {len(results)} rows.") files.download(output_filename)
